一、引言 CATLASS 是 CANN 面向昇腾 NPU 推出的底层向量算子编程库，核心包含 Global Memory（全局内存）、L0/L1/UB 本地缓存多级存储层级。Global Memory 为片外大容量存储，带宽远低于片上缓存，不合理的读写会造成算子性能暴跌。本文围绕 CATLASS 算子标准开发流程，完整实现全局内存分块、连续向量化访存、批量 DMA 搬移、访存合并、地址偏移计算全套代码，解决 Global Memory 随机访问、单次搬移数据量过小、地址不连续带来的带宽浪费问题，适配 910B/310B 系列 NPU。 开发链路：主机侧参数下发→Device 侧 CATLASS 核函数实现 Global Memory 读写→DMA 数据预搬运至 UB 计算→结果回写全局内存，配套编译脚本、性能校验代码。 二、工程目录与编译脚本 build.sh bash 运行 #!/bin/bash # CANN CATLASS Global Memory算子编译脚本 export ASCEND_TOOLKIT=/usr/local/Ascend/ascend-toolkit/latest export PATH=$ASCEND_TOOLKIT/bin:$PATH export LD_LIBRARY_PATH=$ASCEND_TOOLKIT/lib64:$LD_LIBRARY_PATH # 编译Device侧CATLASS内核 atc --kernel=add_gm_kernel.cpp \     --target=ascend910b \     --output=./kernel_bin \     --opt_level=O3 # 编译主机侧调用驱动 g++ main.cpp -o gm_op_run \     -I$ASCEND_TOOLKIT/include \     -L$ASCEND_TOOLKIT/lib64 \     -lascendcl -lcann_ops -lpthread 三、Device 侧 CATLASS 内核：Global Memory 标准访存代码 add_gm_kernel.cpp 核心逻辑：将 Global Memory 大数据分块，使用 DMA 批量连续搬运至 UB 片上缓存，向量计算完成后批量写回全局内存，避免单元素随机读写。 cpp 运行 #include "catlass/catlass.h" using namespace catlass; // 算子入参：全局内存输入输出指针、数据长度 __global__ void add_gm_kernel(GM_ADDR in0_gm, GM_ADDR in1_gm, GM_ADDR out_gm, int data_len) {     // 1. 硬件参数定义：UB单块搬移长度，NPU向量单元大小     const int BLOCK_SIZE = 1024;     const int VEC_NUM = 64;     int total_block = (data_len + BLOCK_SIZE - 1) / BLOCK_SIZE;     // 遍历所有数据分块     for(int blk_idx = 0; blk_idx < total_block; blk_idx++)     {         int blk_start = blk_idx * BLOCK_SIZE;         int cur_len = min(BLOCK_SIZE, data_len - blk_start);         // 2. 在UB上开辟本地缓存，作为Global Memory与计算单元中间缓冲区         UB_BUFFER ub_in0(BLOCK_SIZE);         UB_BUFFER ub_in1(BLOCK_SIZE);         UB_BUFFER ub_out(BLOCK_SIZE);         // 3. DMA批量从Global Memory读入UB，连续地址合并访存         dma_copy(ub_in0, in0_gm + blk_start, cur_len);         dma_copy(ub_in1, in1_gm + blk_start, cur_len);         dma_sync(); // 等待全局内存读取完成         // 4. CATLASS向量计算，在UB高速缓存运算         for(int vec = 0; vec < cur_len; vec += VEC_NUM)         {             int vec_end = min(vec + VEC_NUM, cur_len);             VEC_FLOAT vec0 = ub_in0.load_vec(vec, vec_end);             VEC_FLOAT vec1 = ub_in1.load_vec(vec, vec_end);             VEC_FLOAT vec_res = vec0 + vec1;             ub_out.store_vec(vec_res, vec, vec_end);         }         // 5. DMA批量将计算结果写回Global Memory         dma_copy(out_gm + blk_start, ub_out, cur_len);         dma_sync();     } } Global Memory 关键优化点说明 分块策略：超大全局数组切分为固定 BLOCK_SIZE，单次 DMA 搬运大块连续地址，提升总线带宽利用率； DMA 异步搬运：dma_copy 批量操作替代单元素 gm_load/gm_store，减少 Global Memory 读写系统调用； UB 中转：所有计算在片上 UB 执行，仅首尾与 Global Memory 交互，大幅降低低速片外存储访问次数； 向量化 load/store：CATLASS 向量接口一次性加载 64 个 float，消除循环单元素访存开销。 四、主机侧 CL 运行时调用代码 main.cpp 完成 Host 内存分配、Global Memory 显存申请、内核参数下发、数据拷贝与算子执行，封装完整调用链路。 cpp 运行 #include "ascendcl/acl.h" #include  #include  #define DATA_LEN 16384 int main() {     // 1. 初始化昇腾CL运行时     aclInit(nullptr);     aclrtContext context;     aclrtCreateContext(&context, 0);     aclrtSetCurrentContext(context);     // 2.
